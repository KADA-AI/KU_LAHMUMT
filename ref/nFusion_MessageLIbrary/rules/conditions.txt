# rules/conditions.txt

# 공통 좌표/선/영역/추적 조건
WHEN field=PriorMissionList.CoordinateOrientation IF MissionType==1
WHEN field=PriorMissionList.TargetOrientation     IF MissionType==2

# 0303 UAV 전용
WHEN message=0303 field=AircraftID USE_PROFILE aircraft_id_uav0303_v1

# 편대/편성
WHEN field=FormationInfo REQUIRED_IF IsFormationFlight==1
WHEN field=FormationInfo.leaderAircraftID USE_PROFILE leaderAircraftID_uav_v1

# 웨이포인트/체공
WHEN field=LoiterProperty REQUIRED_IF WaypointPassType==2   # Loiter

# 센서/운용모드별 필드 요구
WHEN field=CoordinateOrientation REQUIRED_IF OperationMode==1
WHEN field=LineSearch            REQUIRED_IF OperationMode==2
WHEN field=AutoTracking          REQUIRED_IF OperationMode==3
WHEN field=AircraftFixed         REQUIRED_IF OperationMode==4
WHEN field=AutoScan              REQUIRED_IF OperationMode==5

# 0303: 경로비행(개별)일 때만 waypointList 사용
WHEN message=0303 field=waypointList REQUIRED_IF IsFormationFlight==0

# 0302: 선행임무일 때만 priorMissionID 필요
WHEN message=0302 field=relatedMission.priorMissionID REQUIRED_IF relatedMissionType==2

# 0401 FlightMode별 조건
WHEN field=LoiterCoordinate   REQUIRED_IF FlightMode==8
WHEN field=TargetFollowingID  REQUIRED_IF FlightMode==9
WHEN field=LeaderAircraftID   REQUIRED_IF FlightMode==6

# 짐벌 제약
CONSTRAINT GimbalYawLimits: leftLimit<=rightLimit

# 0304 Manned 전용 AircraftID 제한
WHEN message=0304 field=AircraftID USE_PROFILE aircraft_id_manned0304_v1
